// Status Badge Component
function StatusBadge({ status }: { status: StatusType }) {
  const styles: Record<StatusType, { bg: string; text: string; border: string; icon: string }> = {
    Revisi: { bg: 'bg-amber-50', text: 'text-amber-700', border: 'border-amber-200', icon: '⟳' },
    Relevan: { bg: 'bg-emerald-50', text: 'text-emerald-700', border: 'border-emerald-200', icon: '✓' },
    Obsolete: { bg: 'bg-red-50', text: 'text-red-700', border: 'border-red-200', icon: '⊘' },
    Menunggu: { bg: 'bg-blue-50', text: 'text-blue-700', border: 'border-blue-200', icon: '⏳' },
    '-': { bg: 'bg-slate-50', text: 'text-slate-600', border: 'border-slate-200', icon: '?' },
  };
  const { bg, text, border, icon } = styles[status] || styles['-'];
  return (
    <span className={`inline-flex items-center gap-1.5 px-3 py-1.5 rounded-lg text-sm font-semibold whitespace-nowrap border ${bg} ${text} ${border}`}>
      {icon} {status}
    </span>
  );
}

// Email Status Badge Component
function EmailStatusBadge({ status }: { status: string }) {
  const styles: Record<string, { bg: string; text: string; border: string; icon: string }> = {
    pending: { bg: 'bg-amber-50', text: 'text-amber-700', border: 'border-amber-200', icon: '⏳' },
    sent: { bg: 'bg-emerald-50', text: 'text-emerald-700', border: 'border-emerald-200', icon: '✓' },
    failed: { bg: 'bg-red-50', text: 'text-red-700', border: 'border-red-200', icon: '✗' },
    cancelled: { bg: 'bg-slate-50', text: 'text-slate-700', border: 'border-slate-200', icon: '⊘' },
  };
  const { bg, text, border, icon } = styles[status] || styles['pending'];
  return (
    <span className={`inline-flex items-center gap-1 px-2 py-0.5 rounded text-xs font-semibold border ${bg} ${text} ${border}`}>
      {icon} {status.toUpperCase()}
    </span>
  );
}

// Add Procedure Modal Component
function AddProcedureModal({
  isOpen,
  onClose,
  onSave,
  value,
  onChange,
  isSaving,
}: {
  isOpen: boolean;
  onClose: () => void;
  onSave: () => void;
  value: Omit<ProcedureDocument, 'status' | 'revisi' | 'id'> & { revisi?: string };
  onChange: (field: keyof Omit<ProcedureDocument, 'status' | 'revisi' | 'id'>, val: string) => void;
  isSaving: boolean;
}) {
  if (!isOpen) return null;

  const handleSave = () => {
    if (
      !value.namaDokumen ||
      !value.nomorDokumen ||
      !value.levelDokumen ||
      !value.tanggalSelesaiApproval ||
      !value.owner ||

      !value.email
    ) {
      alert('Semua kolom wajib diisi.');
      return;

  };

  return (
    <div className="fixed inset-0 bg-gray-700/60 bg-opacity-50 backdrop-blur-sm z-50 flex items-center justify-center p-4">
      <div className="bg-white rounded-xl w-full max-w-2xl shadow-2xl">

        </div>

        <div className="p-6 grid grid-cols-2 gap-4">
          {([
            { label: 'Nama Dokumen', key: 'namaDokumen' },
            { label: 'Nomor Dokumen', key: 'nomorDokumen' },
            { label: 'Level Dokumen', key: 'levelDokumen' },
            { label: 'Owner', key: 'owner' },
            { label: 'PIC', key: 'pic' },

// Email Schedule Modal Component
function EmailScheduleModal({
  isOpen,
  onClose,
  onSave,
  nomorDokumen,
  currentSchedule,
  isSaving,
}: {
  isOpen: boolean;
  onClose: () => void;
  onSave: (nomorDokumen: string, scheduledEmailAt: string) => Promise<void>;
  nomorDokumen: string;
  currentSchedule?: EmailSchedule;
  isSaving: boolean;
}) {
  const [scheduledDate, setScheduledDate] = useState('');
  const [scheduledTime, setScheduledTime] = useState('');
  const [error, setError] = useState('');

  useEffect(() => {
    if (currentSchedule) {
      const date = new Date(currentSchedule.scheduledEmailAt);
      setScheduledDate(date.toISOString().split('T')[0]);
      setScheduledTime(date.toTimeString().slice(0, 5));
    } else {
      // Default: 7 hari dari sekarang, jam 16:00
      const defaultDate = new Date();
      defaultDate.setDate(defaultDate.getDate() + 7);
      defaultDate.setHours(16, 0, 0, 0);
      setScheduledDate(defaultDate.toISOString().split('T')[0]);
      setScheduledTime('16:00');
    }
    setError('');
  }, [currentSchedule, isOpen]);

  if (!isOpen) return null;

  const handleSave = async () => {
    if (!scheduledDate || !scheduledTime) {
      setError('Tanggal dan waktu harus diisi');
      return;
    }

    const scheduledDateTime = new Date(`${scheduledDate}T${scheduledTime}`);
    const now = new Date();

    if (scheduledDateTime < now) {
      setError('Tanggal/waktu tidak boleh di masa lalu');
      return;
    }

    try {
      await onSave(nomorDokumen, scheduledDateTime.toISOString());
      onClose();
    } catch (err: any) {
      setError(err.message || 'Gagal menyimpan jadwal');
    }
  };

  // Get minimum date (today)
  const minDate = new Date().toISOString().split('T')[0];

  return (
    <div className="fixed inset-0 bg-gray-700/60 bg-opacity-50 backdrop-blur-sm z-50 flex items-center justify-center p-4">
      <div className="bg-white rounded-xl w-full max-w-md shadow-2xl">
        <div className="px-6 py-4 border-b border-slate-200 bg-gradient-to-r from-blue-50 to-blue-100">
          <h3 className="text-lg font-bold text-slate-900">
            {currentSchedule ? 'Edit Jadwal Email' : 'Atur Jadwal Email'}
          </h3>
          <p className="text-xs text-slate-600 mt-1">
            Dokumen: {nomorDokumen}
          </p>
        </div>

        <div className="p-6 space-y-4">
          <div>
            <label className="block text-sm font-semibold text-slate-700 mb-2">
              Tanggal Kirim
            </label>
            <input
              type="date"
              value={scheduledDate}
              onChange={(e) => setScheduledDate(e.target.value)}
              min={minDate}
              className="w-full px-4 py-2.5 border border-slate-300 rounded-lg text-sm text-slate-900 focus:outline-none focus:ring-2 focus:ring-blue-500 focus:border-transparent transition-all"
              disabled={isSaving}
            />
          </div>

          <div>
            <label className="block text-sm font-semibold text-slate-700 mb-2">
              Waktu Kirim
            </label>
            <input
              type="time"
              value={scheduledTime}
              onChange={(e) => setScheduledTime(e.target.value)}
              className="w-full px-4 py-2.5 border border-slate-300 rounded-lg text-sm text-slate-900 focus:outline-none focus:ring-2 focus:ring-blue-500 focus:border-transparent transition-all"
              disabled={isSaving}
            />
          </div>

          {currentSchedule && (
            <div className="bg-slate-50 rounded-lg p-3 text-xs">
              <div className="flex justify-between mb-1">
                <span className="text-slate-600">Status:</span>
                <EmailStatusBadge status={currentSchedule.emailStatus} />
              </div>
              {currentSchedule.lastSentAt && (
                <div className="flex justify-between">
                  <span className="text-slate-600">Terakhir dikirim:</span>
                  <span className="text-slate-900">
                    {new Date(currentSchedule.lastSentAt).toLocaleString('id-ID')}
                  </span>
                </div>
              )}
              {currentSchedule.retryCount > 0 && (
                <div className="flex justify-between">
                  <span className="text-slate-600">Retry:</span>
                  <span className="text-slate-900">{currentSchedule.retryCount}x</span>
                </div>
              )}
            </div>
          )}

          {error && (
            <div className="bg-red-50 border border-red-200 text-red-700 px-4 py-3 rounded-lg text-sm">
              {error}
            </div>
          )}
        </div>

        <div className="px-6 py-4 border-t border-slate-200 bg-slate-50 flex justify-end gap-3">
          <button
            onClick={onClose}
            className="px-5 py-2.5 text-sm font-medium text-slate-700 hover:bg-slate-200 rounded-lg transition-all active:scale-95"
            disabled={isSaving}
          >
            Batal
          </button>
          <button
            onClick={handleSave}
            className="px-5 py-2.5 bg-blue-600 hover:bg-blue-700 text-white text-sm font-semibold rounded-lg transition-all hover:shadow-md active:scale-95 disabled:opacity-50 disabled:cursor-not-allowed flex items-center gap-2"
            disabled={isSaving}
          >
            {isSaving && (
              <svg className="animate-spin h-4 w-4 text-white" xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24">
                <circle className="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" strokeWidth="4"></circle>
                <path className="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.938l3-2.647z"></path>
              </svg>
            )}
            {isSaving ? 'Menyimpan...' : 'Simpan Jadwal'}
          </button>
        </div>
      </div>
    </div>
  );
}

// Helper function to format date
function formatToDDMonYYYY(iso: string): string {
  const d = new Date(iso);
  if (isNaN(d.getTime())) return iso;
  const MONTH_NAMES = ['Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec'];
  return `${d.getDate()}-${MONTH_NAMES[d.getMonth()]}-${d.getFullYear()}`;
}
